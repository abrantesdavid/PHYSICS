\section*{Vetores e componentes}
\begin{align*}
\vec A &= A_x\vi+A_y\vj+A_z\vk,\\
A_x &= A\cos\theta, & A_y &= A\sin\theta,\\
A &= \norm{\vec A}=\sqrt{A_x^2+A_y^2+A_z^2},\\
\tan\theta &= \frac{A_y}{A_x}, &
\theta &= \operatorname{atan2}(A_y,A_x).
\end{align*}

\section*{Operações vetoriais}
\begin{align*}
\vec A+\vec B
&=(A_x+B_x)\vi+(A_y+B_y)\vj+(A_z+B_z)\vk,\\
\vec A-\vec B
&=(A_x-B_x)\vi+(A_y-B_y)\vj+(A_z-B_z)\vk,\\
\lambda\vec A
&=(\lambda A_x)\vi+(\lambda A_y)\vj+(\lambda A_z)\vk,\\
\norm{\lambda\vec A}&=|\lambda|\norm{\vec A},\\
\hat a&=\frac{\vec A}{\norm{\vec A}},\qquad \vec A\ne\vec0,\\
\vec A+\vec B+\vec C&=\vec0
\quad\Longrightarrow\quad \vec C=-(\vec A+\vec B).
\end{align*}

\section*{Produto escalar, ângulo e projeção}
\begin{align*}
\vec A\cdot\vec B
&=A_xB_x+A_yB_y+A_zB_z,\\
\vec A\cdot\vec B&=AB\cos\theta,\\
\theta&=\arccos\left(\frac{\vec A\cdot\vec B}{AB}\right),\\
\vec A\perp\vec B&\Longleftrightarrow \vec A\cdot\vec B=0,\\
\operatorname{comp}_{\vec B}\vec A
&=\frac{\vec A\cdot\vec B}{\norm{\vec B}},\\
\operatorname{proj}_{\vec B}\vec A
&=\frac{\vec A\cdot\vec B}{\norm{\vec B}^{2}}\vec B.
\end{align*}

\section*{Produto vetorial e torque}
\begin{align*}
\norm{\vec A\times\vec B}&=AB\sin\theta,\\
\vec A\times\vec B
&=\begin{vmatrix}
\vi&\vj&\vk\\ A_x&A_y&A_z\\ B_x&B_y&B_z
\end{vmatrix},\\
\vec A\times\vec B
&=(A_yB_z-A_zB_y)\vi+(A_zB_x-A_xB_z)\vj\\
&\quad +(A_xB_y-A_yB_x)\vk,\\
\vec B\times\vec A&=-(\vec A\times\vec B),\\
\vec A\parallel\vec B&\Longleftrightarrow \vec A\times\vec B=\vec0,\\
\vi\times\vj&=\vk, & \vj\times\vk&=\vi, & \vk\times\vi&=\vj,\\
\vec\tau&=\vec r\times\vec F,\\
\tau&=rF\sin\theta,\\
\tau_z&=r_xF_y-r_yF_x.
\end{align*}

\section*{Equilíbrio}
\begin{align*}
\sum\vec F&=m\vec a,\\
\sum\vec F&=\vec0,\\
\sum F_x&=0, & \sum F_y&=0, & \sum F_z&=0,\\
\vec F_3&=-(\vec F_1+\vec F_2).
\end{align*}

\section*{Posição, deslocamento, velocidade e aceleração}
\begin{align*}
\vec r(t)&=x(t)\vi+y(t)\vj+z(t)\vk,\\
\Delta\vec r&=\vec r_f-\vec r_i,\\
\vec v_m&=\frac{\Delta\vec r}{\Delta t}
=\frac{\vec r_f-\vec r_i}{t_f-t_i},\\
\vec v(t)&=\lim_{\Delta t\to0}
\frac{\vec r(t+\Delta t)-\vec r(t)}{\Delta t},\\
\vec v(t)&=\frac{d\vec r}{dt},\\
v(t)&=\norm{\vec v(t)}=\sqrt{v_x^2+v_y^2+v_z^2},\\
\vec a_m&=\frac{\Delta\vec v}{\Delta t},\\
\vec a(t)&=\frac{d\vec v}{dt}=\frac{d^2\vec r}{dt^2}.
\end{align*}

\section*{Aceleração constante}
\begin{align*}
\vec v(t)&=\vec v_0+\vec a,t,\\
\vec r(t)&=\vec r_0+\vec v_0t+\frac12\vec a,t^2,\\
v_x(t)&=v_{0x}+a_xt, & v_y(t)&=v_{0y}+a_yt,\\
x(t)&=x_0+v_{0x}t+\frac12a_xt^2,\\
y(t)&=y_0+v_{0y}t+\frac12a_yt^2,\\
v^2&=v_0^2+2a\Delta s.
\end{align*}

\section*{Lançamento oblíquo}
\begin{align*}
v_{0x}&=v_0\cos\theta, & v_{0y}&=v_0\sin\theta,\\
\vec a&=-g\vj,\\
x(t)&=x_0+v_0\cos\theta\,t,\\
y(t)&=y_0+v_0\sin\theta\,t-\frac12gt^2,\\
v_x(t)&=v_0\cos\theta,\\
v_y(t)&=v_0\sin\theta-gt,\\
\vec v(t)&=v_0\cos\theta\,\vi+
\left(v_0\sin\theta-gt\right)\vj,\\
v(t)&=\sqrt{v_x^2+v_y^2},\\
\phi(t)&=\operatorname{atan2}(v_y,v_x),\\
t_{\mathrm{subida}}&=\frac{v_0\sin\theta}{g},\\
H_{\max}-y_0&=\frac{v_0^2\sin^2\theta}{2g},\\
T&=\frac{2v_0\sin\theta}{g},\\
L&=v_0\cos\theta\,T
=\frac{v_0^2\sin(2\theta)}{g},\\
y-y_0&=(x-x_0)\tan\theta
-\frac{g(x-x_0)^2}{2v_0^2\cos^2\theta}.
\end{align*}

\section*{Queda livre}
\begin{align*}
\vec a&=-g\vj,\\
y(t)&=y_0+v_{0y}t-\frac12gt^2,\\
v_y(t)&=v_{0y}-gt,\\
v_y^2&=v_{0y}^2-2g(y-y_0),\\
v_{0y}=0&\quad\Longrightarrow\quad y(t)=y_0-\frac12gt^2.
\end{align*}

\section*{Movimento circular uniforme}
\begin{align*}
\theta(t)&=\theta_0+\omega t,\\
\vec r(t)&=R\cos(\theta_0+\omega t)\vi
+R\sin(\theta_0+\omega t)\vj,\\
\vec v(t)&=-R\omega\sin(\theta_0+\omega t)\vi
+R\omega\cos(\theta_0+\omega t)\vj,\\
v&=R|\omega|,\\
\vec a(t)&=-\omega^2\vec r(t),\\
a_c&=R\omega^2=\frac{v^2}{R},\\
\vec r\cdot\vec v&=0,\\
T&=\frac{2\pi}{|\omega|}, & f&=\frac1T, & |\omega|&=2\pi f.
\end{align*}

\section*{Velocidade relativa}
\begin{align*}
\vec r_{A/C}&=\vec r_{A/B}+\vec r_{B/C},\\
\vec v_{A/C}&=\vec v_{A/B}+\vec v_{B/C},\\
\vec v_{A/B}&=\vec v_{A/C}-\vec v_{B/C},\\
t_{\mathrm{travessia}}&=\frac{d}{v_{\perp}},\\
\Delta x_{\mathrm{deriva}}&=v_{\parallel}t_{\mathrm{travessia}},\\
v_{\mathrm{resultante}}&=\sqrt{v_{\perp}^{2}+v_{\parallel}^{2}},\\
\tan\alpha&=\frac{v_{\parallel}}{v_{\perp}}.
\end{align*}
